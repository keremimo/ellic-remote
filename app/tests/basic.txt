# Melodee Remote in the host simulator: ./build.sh --host, then
#   app/build/host/melodee-remote_host app/tests/basic.txt app/build/tests
# Its own screen, the keys and the panel as MIDI, Melodee claiming it over USB and Bluetooth and letting it go.
wait 1300
expect own 1
expect host 0
text title MELODEE REMOTE
text line0 WAITING
shot waiting.png
# the keys play notes on channel 1, F3 = 53; the buttons and encoders go out on channel 16
key 7 down
wait 30
expect_out USB 90 3C 64
key 7 up
wait 30
expect_out USB 80 3C 00
tap FX
wait 30
expect_out USB 9F 00 7F
expect_out USB 8F 00 00
turn K1 2
wait 30
expect_out USB BF 13 41
turn SELECT -1
wait 30
expect_out USB BF 10 3F
out
# OCT+ moves the keys (and still goes out as a button); both back to 0
tap OCT+
wait 30
expect octave 1
expect_out USB 9F 0D 7F
key 0 down
wait 30
expect_out USB 90 41 64
key 0 up
wait 30
press OCT-
press OCT+
wait 30
release OCT-
release OCT+
wait 30
expect octave 0
expect setup 0
out
# Melodee over USB: HELLO claims it; the FM-1 answers with its own and with MASTER
sysex F0 7D 4D 52 01 01 F7
wait 30
expect host 1
expect own 0
expect frames 1
expect_out USB F0 7D 4D
expect_out USB BF 07 3F
out
